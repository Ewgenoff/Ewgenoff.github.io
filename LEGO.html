// Константы столбцов таблицы (начиная с 0) const COL_SET_NUM = 0; // Столбец A (Номер) const COL_NAME = 2; // Столбец C (Название) const COL_PIECES = 3; // Столбец D (Кол-во деталей) const COL_YEAR = 4; // Столбец E (Год) const COL_THEME = 5; // Столбец F (Серия) const COL_STATUS = 7; // Столбец H (Статус) // Функция получения ссылки на изображение по номеру набора function getSetImageUrl(setNumber) { if (!setNumber) return 'https://via.placeholder.com/300x220?text=No+Image'; let cleanNum = setNumber.toString().trim(); // Если в номере нет тире (например, "75301"), добавляем "-1" для CDN Rebrickable if (!cleanNum.includes('-')) { cleanNum = `${cleanNum}-1`; } return `https://cdn.rebrickable.com/media/sets/${cleanNum}.jpg`; } // Рендеринг карточки набора function createSetCard(row) { const setNum = row[COL_SET_NUM] || '—'; const name = row[COL_NAME] || 'Без названия'; const pieces = row[COL_PIECES] || '0'; const year = row[COL_YEAR] || '—'; const theme = row[COL_THEME] || 'Разное'; const status = row[COL_STATUS] || 'Не указан'; const imageUrl = getSetImageUrl(setNum); return `
${status}
№ ${setNum}

${name}

${theme} ${year} 🧩 ${pieces} дет.
`; } // Отображение в модальном окне описания function showSetDetails(row) { const theme = row[COL_THEME] || '—'; const pieces = row[COL_PIECES] || '—'; document.getElementById('modal-theme').textContent = theme; document.getElementById('modal-pieces').textContent = `${pieces} деталей`; }
